Execute Program 1 then follow the below comments (AWS EC2 Linux instance and Connect using SSH)

# Step 1: Update package information
sudo apt update

# Step 2: Upgrade installed packages
sudo apt upgrade -y

# Step 3: Start Docker service
sudo systemctl start docker

#Step 4: Install Docker Service
sudo apt install docker.io -y

# Step 5: Enable Docker at system startup
sudo systemctl enable docker

# Step 6: Check Docker version
docker --version

# Step 7: Check Docker service status
sudo systemctl status docker

# Step 8: Add Ubuntu user to Docker group
sudo usermod -aG docker ubuntu

# Step 9: Refresh group membership
newgrp docker

# Step 10: Test Docker installation
docker run hello-world

# Step 11: List Docker images
docker images

# Step 12: List all Docker containers
docker ps -a
